"use client";

import Link from "next/link";
import SavedPlacedParlaysLedgerPanel from "./SavedPlacedParlaysLedgerPanel";
import { useEffect, useMemo, useState } from "react";

export default function ParlayResults({
  parlays,
  counts,
  savedPlacedParlays = [],
  savedLegUsageMap,
  onSavePlacedParlay,
  onSavePlaceabilityDecision,
  onUpdateParsedRowOddsFromParlayCard,
  onClearSavedParlays,
  onDeleteSavedParlay,
  onUpdateSavedParlay,
  onConfirmSavedParlayPlaced,
  onSetSavedParlayResult,
  formatSavedDateTime,
  boostWallet = [],
  blockedParlayLegs = [],
  onBlockParlayLeg,
  onUnblockParlayLeg,
  selectedDevigMethod = "power",
}) {
  const [sectionCollapsed, setSectionCollapsed] = useState(false);
  const [converterInput, setConverterInput] = useState("");
  const [converterResult, setConverterResult] = useState(null);
  const [collapsedMap, setCollapsedMap] = useState({});
  const [selectedBoostByParlayId, setSelectedBoostByParlayId] = useState({});
  const [stakeByParlayId, setStakeByParlayId] = useState({});
  const [showRecommendedDevigByParlayId, setShowRecommendedDevigByParlayId] = useState({});
  const [actualSgpOddsByParlayId, setActualSgpOddsByParlayId] = useState({});
  const [sgpOddsModeByParlayId, setSgpOddsModeByParlayId] = useState({});
  const [legOddsByParlayId, setLegOddsByParlayId] = useState({});
  const [fairOddsByParlayId, setFairOddsByParlayId] = useState({});  const [placeabilityByParlayId, setPlaceabilityByParlayId] = useState({});
  const [placeabilityNotesByParlayId, setPlaceabilityNotesByParlayId] = useState({});

  const safeParlays = Array.isArray(parlays) ? parlays : [];
  const parlayResultSignature = useMemo(
    () =>
      safeParlays
        .map((parlay) =>
          [
            parlay.id,
            parlay.rawParlayAmerican,
            parlay.boostedParlayAmerican,
            ...(parlay.legs || []).map((leg) =>
              [
                leg.eventName,
                leg.marketType,
                leg.subjectName,
                leg.selectionLabel,
                leg.lineValue,
                leg.oddsAmerican,
                leg.fairAmerican,
              ].join("|")
            ),
          ].join("::")
        )
        .join("||||"),
    [safeParlays]
  );

  useEffect(() => {
    setCollapsedMap({});
  }, [parlayResultSignature]);

  const safeSavedPlacedParlays = repairMojibakeDeep(savedPlacedParlays);
  function toggleParlay(id) {
    setCollapsedMap((prev) => ({
      ...prev,
      [id]: !prev[id],
    }));
  }

  function toggleRecommendedDevig(id) {
    setShowRecommendedDevigByParlayId((prev) => ({
      ...prev,
      [id]: !prev[id],
    }));
  }

  function handleBoostSelect(parlay, boostId) {
    setSelectedBoostByParlayId((prev) => ({
      ...prev,
      [parlay.id]: boostId,
    }));

    const selectedBoost = (boostWallet || []).find((boost) => boost.id === boostId);
    const currentStake = stakeByParlayId[parlay.id];

    if (
      selectedBoost &&
      (currentStake === undefined || currentStake === null || currentStake === "") &&
      Number.isFinite(Number(selectedBoost.maxStake))
    ) {
      setStakeByParlayId((prev) => ({
        ...prev,
        [parlay.id]: Number(selectedBoost.maxStake),
      }));
    }
  }

  function handlePlaceabilityDecisionChange(parlay, nextStatus, nextNotes, context = {}) {
    const status = String(nextStatus || "not_checked").trim();
    const notes = String(nextNotes || "");

    // Do not create a saved record for the untouched default state.
    if (status === "not_checked" && !notes.trim()) return;

    const actualSgpRegrade = context.actualSgpRegrade || null;

    onSavePlaceabilityDecision?.(parlay, {
      boostId: context.selectedBoostId || selectedBoostByParlayId[parlay.id] || "",
      placedStake: Number(context.stakeValue || parlay?.stake || 0),
      placedOddsAmerican:
        actualSgpRegrade?.actualBoostedAmerican ??
        parlay?.boostedParlayAmerican ??
        parlay?.rawParlayAmerican,
      actualBookSgpOddsAmerican: actualSgpRegrade?.actualRawAmerican ?? null,
      actualBoostedSgpOddsAmerican: actualSgpRegrade?.actualBoostedAmerican ?? null,
      actualRawEvPct: actualSgpRegrade?.actualRawEvPct ?? null,
      actualBoostedEvPct: actualSgpRegrade?.actualBoostedEvPct ?? null,
      actualExpectedProfitAtStake: actualSgpRegrade?.actualExpectedProfitAtStake ?? null,
      actualSuggestedKellyStake: actualSgpRegrade?.actualSuggestedKellyStake ?? null,
      actualRawSuggestedKellyStake: actualSgpRegrade?.actualRawSuggestedKellyStake ?? null,
      actualBoostedSuggestedKellyStake: actualSgpRegrade?.actualBoostedSuggestedKellyStake ?? null,
      actualGradeTier: actualSgpRegrade?.actualGradeTier ?? "",
      actualPlayLabel: actualSgpRegrade?.actualPlayLabel ?? "",
      legOddsOverrides: legOddsByParlayId[parlay.id] || {},
      placeabilityStatus: status,
      placeabilityNotes: notes,
    });
  }

  return (
    <section style={sectionStyle}>
      <div style={sectionHeaderStyle}>
        <div>
          <h2 style={h2Style}>8. Parlay Results</h2>
          <div style={subtleStyle}>
            Current devig method: <strong>{formatDevigMethodLabel(selectedDevigMethod)}</strong>
          </div>
        </div>

        <button
          type="button"
          onClick={() => setSectionCollapsed((prev) => !prev)}
          style={toggleButtonStyle}
        >
          {sectionCollapsed ? "Show Parlay Results" : "Collapse Parlay Results"}
        </button>
      </div>

      <div style={countsRowStyle}>
        <span style={countPillStyle}>Eligible Markets: {counts?.eligibleMarkets ?? 0}</span>
        <span style={countPillStyle}>Eligible Legs: {counts?.eligibleLegs ?? 0}</span>
        <span style={countPillStyle}>Generated Combos: {counts?.generatedCombos ?? 0}</span>
      </div>

      {sectionCollapsed ? (
        <div style={collapsedNoticeStyle}>
          Results are hidden. Counts above still update when filters change.
        </div>
      ) : (
        <>
          {blockedParlayLegs.length ? (
            <div style={blockedLegsPanelStyle}>
              <div style={blockedLegsTitleStyle}>Blocked Legs</div>
              <div style={blockedLegsSubtleStyle}>
                These legs are manually excluded from suggested parlay calculations.
              </div>

              <div style={blockedLegsListStyle}>
                {blockedParlayLegs.map((blocked) => (
                  <div key={blocked.id} style={blockedLegPillStyle}>
                    <span>{blocked.displayLabel || formatBlockedLegLabel(blocked)}</span>
                    <button
                      type="button"
                      onClick={() => onUnblockParlayLeg?.(blocked.id)}
                      style={removeBlockedLegButtonStyle}
                    >
                      Remove
                    </button>
                  </div>
                ))}
              </div>
            </div>
          ) : null}

          {counts?.rejections ? (
            <div style={rejectionsRowStyle}>
              <span style={rejectionPillStyle}>No Fair Odds: {counts.rejections.noFairOdds ?? 0}</span>
              <span style={rejectionPillStyle}>No Target Quote: {counts.rejections.noTargetQuote ?? 0}</span>
              <span style={rejectionPillStyle}>Below Leg Threshold: {counts.rejections.belowLegThreshold ?? 0}</span>
              <span style={rejectionPillStyle}>Same-Sport Blocked: {counts.rejections.sameSportBlocked ?? 0}</span>
              <span style={rejectionPillStyle}>Same-Game Blocked: {counts.rejections.sameGameBlocked ?? 0}</span>
              <span style={rejectionPillStyle}>Repeats Blocked: {counts.rejections.repeatsBlocked ?? 0}</span>
              <span style={rejectionPillStyle}>Manual Blocks: {counts.rejections.manualBlocked ?? 0}</span>
              <span style={rejectionPillStyle}>Non-Positive EV: {counts.rejections.nonPositiveParlayEv ?? 0}</span>
              <span style={rejectionPillStyle}>Market Mode Filtered: {counts.rejections.filteredByMarketMode ?? 0}</span>
              <span style={rejectionPillStyle}>Extreme Odds Filtered: {counts.rejections.filteredByExtremeOdds ?? 0}</span>
            </div>
          ) : null}

          <div style={converterBlockStyle}>
            <div style={converterLabelStyle}>Odds Converter</div>
            <div style={converterWrapStyle}>
              <input
                value={converterInput}
                onChange={(e) => {
                  const val = e.target.value;
                  setConverterInput(val);
                  setConverterResult(convertOdds(val));
                }}
                placeholder="+150 or 2.50"
                style={converterInputStyle}
              />

              <div style={converterResultStyle}>
                {converterResult ? (
                  converterResult.type === "american_to_decimal" ? (
                    <>Decimal: {converterResult.decimal.toFixed(3)}</>
                  ) : (
                    <>American: {formatAmerican(converterResult.american)}</>
                  )
                ) : (
                  "—"
                )}
              </div>
            </div>
          </div>

          {safeParlays.length === 0 ? (
            <div style={emptyWarningStyle}>No positive EV parlays found with available odds.</div>
          ) : (
            <div style={{ display: "grid", gap: 12 }}>
              {safeParlays.map((parlay, idx) => {
                const isCollapsed = !!collapsedMap[parlay.id];
                const showRecommended = !!showRecommendedDevigByParlayId[parlay.id];
                const selectedBoostId = selectedBoostByParlayId[parlay.id] || "";
                const stakeValue = stakeByParlayId[parlay.id] ?? "";
                const actualSgpOddsText = actualSgpOddsByParlayId[parlay.id] ?? "";
                const sgpOddsMode =
                  sgpOddsModeByParlayId[parlay.id] ||
                  (getSameGameClusterInfo(parlay).isMixedCluster ? "cluster" : "full");
                const actualSgpRegrade = buildActualSgpRegrade(parlay, actualSgpOddsText, stakeValue, {
                  sgpOddsMode,
                });
                const placeabilityStatus = placeabilityByParlayId[parlay.id] || "not_checked";
                const placeabilityNotes = placeabilityNotesByParlayId[parlay.id] || "";
                const uniqueEventNames = Array.from(
                  new Set((parlay.legs || []).map((leg) => String(leg.eventName || "").trim()).filter(Boolean))
                );
                const sameGameClusterInfo = getSameGameClusterInfo(parlay);
                const isSameGameCandidate = (parlay.legs || []).length > 1 && uniqueEventNames.length === 1;
                const hasSgpPricingToolbox = sameGameClusterInfo.hasCluster;
                const displayGradeTier = actualSgpRegrade?.actualGradeTier || parlay.gradeTier;
                const displayPlayLabel = actualSgpRegrade?.actualPlayLabel || parlay.playLabel;
                const displayExpectedValuePct = actualSgpRegrade?.actualBoostedEvPct ?? parlay.expectedValuePct;
                const displayBoostedAmerican = actualSgpRegrade?.actualBoostedAmerican ?? parlay.boostedParlayAmerican;
                const displayExpectedProfitAtStake = actualSgpRegrade?.actualExpectedProfitAtStake ?? parlay.expectedProfitAtStake;
                const displayBoostedKellyStake = actualSgpRegrade?.actualBoostedSuggestedKellyStake ?? parlay.boostedSuggestedKellyStake ?? parlay.suggestedKellyStake ?? 0;
                const displayRawKellyStake = actualSgpRegrade?.actualRawSuggestedKellyStake ?? parlay.rawSuggestedKellyStake ?? 0;

                return (
                  <div key={parlay.id} style={cardStyle}>
                    <div style={cardHeaderStyle}>
                      <div>
                        <div style={{ fontWeight: 800 }}>
                          Candidate #{idx + 1} — {displayGradeTier} / {displayPlayLabel}
                          {actualSgpRegrade ? (
                            <span style={actualRegradeBadgeStyle}>Regraded</span>
                          ) : null}
                        </div>
                        <div style={subtleStyle}>
                          EV {formatPct(displayExpectedValuePct)} • Boosted {formatAmerican(displayBoostedAmerican)} • Devig {parlay.devigMethodLabel || formatDevigMethodLabel(selectedDevigMethod)}
                        </div>
                      </div>

                      <div style={saveControlWrapStyle}>
                        <label style={miniLabelStyle}>
                          Attach Boost
                          <select
                            value={selectedBoostId}
                            onChange={(event) => handleBoostSelect(parlay, event.target.value)}
                            style={boostSelectStyle}
                          >
                            <option value="">No boost attached</option>
                            {boostWallet
                              .filter((boost) => boost.status !== "used" && boost.status !== "expired")
                              .map((boost) => (
                                <option key={boost.id} value={boost.id}>
                                  {boost.sportsbook} — {boost.name} ({boost.boostPct}%)
                                </option>
                              ))}
                          </select>
                        </label>

                        <label style={miniLabelStyle}>
                          Stake to Save
                          <input
                            type="number"
                            value={stakeValue}
                            placeholder={String(parlay.stake ?? "Stake")}
                            onChange={(event) =>
                              setStakeByParlayId((prev) => ({
                                ...prev,
                                [parlay.id]: event.target.value,
                              }))
                            }
                            style={stakeInputStyle}
                          />
                        </label>

                        <button
                          type="button"
                          onClick={() => {
                            if (isRejectedPlaceabilityStatus(placeabilityStatus)) {
                              handlePlaceabilityDecisionChange(parlay, placeabilityStatus, placeabilityNotes, {
                                selectedBoostId,
                                stakeValue,
                                actualSgpRegrade,
                              });
                              return;
                            }

                            onSavePlacedParlay?.(parlay, {
                              boostId: selectedBoostId,
                              placedStake: Number(stakeValue || parlay.stake || 0),
                              placedOddsAmerican:
                                actualSgpRegrade?.actualBoostedAmerican ??
                                parlay.boostedParlayAmerican ??
                                parlay.rawParlayAmerican,
                              actualBookSgpOddsAmerican: actualSgpRegrade?.actualRawAmerican ?? null,
                              actualBoostedSgpOddsAmerican: actualSgpRegrade?.actualBoostedAmerican ?? null,
                              actualRawEvPct: actualSgpRegrade?.actualRawEvPct ?? null,
                              actualBoostedEvPct: actualSgpRegrade?.actualBoostedEvPct ?? null,
                              actualExpectedProfitAtStake: actualSgpRegrade?.actualExpectedProfitAtStake ?? null,
                              actualSuggestedKellyStake: actualSgpRegrade?.actualSuggestedKellyStake ?? null,
                              actualRawSuggestedKellyStake: actualSgpRegrade?.actualRawSuggestedKellyStake ?? null,
                              actualBoostedSuggestedKellyStake: actualSgpRegrade?.actualBoostedSuggestedKellyStake ?? null,
                              actualGradeTier: actualSgpRegrade?.actualGradeTier ?? "",
                              actualPlayLabel: actualSgpRegrade?.actualPlayLabel ?? "",
                              legOddsOverrides: legOddsByParlayId[parlay.id] || {},
                              placeabilityStatus,
                              placeabilityNotes,
                            });
                          }}
                          style={savePlacedButtonStyle}
                        >
                          {isRejectedPlaceabilityStatus(placeabilityStatus) ? "Save Rejection Only" : "Save Candidate"}
                        </button>


                        <button type="button" onClick={() => toggleParlay(parlay.id)} style={toggleButtonStyle}>
                          {isCollapsed ? "Show" : "Hide"}
                        </button>

                        <span style={subtleStyle}>Saved candidates do not count in P&L until Confirm Placed.</span>
                      </div>
                    </div>

                    {!isCollapsed && (
                      <>
                        <div style={parlayActionRowStyle}>
                          <Link href={buildToolsLink(parlay)} style={toolsLinkStyle}>
                            Open in Tools
                          </Link>
                        </div>

                        {hasSgpPricingToolbox ? (
                          <div style={sgpOverrideBoxStyle}>
                            <div style={sgpWarningStyle}>
                              {isSameGameCandidate
                                ? "SGP odds are estimated by multiplying leg odds. Same-game parlays are book-priced and correlated. Enter the actual book SGP odds BEFORE any profit boost."
                                : "This parlay has a same-game cluster. Choose whether the odds you enter are the same-game cluster price only or the full parlay price. Enter book odds BEFORE any profit boost."}
                            </div>

                            <div style={sgpGridStyle}>
                              {sameGameClusterInfo.isMixedCluster ? (
                                <label style={miniLabelStyle}>
                                  Odds Entered Are
                                  <select
                                    value={sgpOddsMode}
                                    onChange={(event) =>
                                      setSgpOddsModeByParlayId((prev) => ({
                                        ...prev,
                                        [parlay.id]: event.target.value,
                                      }))
                                    }
                                    style={boostSelectStyle}
                                  >
                                    <option value="cluster">Same-game cluster only</option>
                                    <option value="full">Full parlay odds</option>
                                  </select>
                                </label>
                              ) : null}

                              <label style={miniLabelStyle}>
                                Actual Book SGP Odds (before boost)
                                <input
                                  value={actualSgpOddsText}
                                  placeholder="+425, not boosted"
                                  onChange={(event) =>
                                    setActualSgpOddsByParlayId((prev) => ({
                                      ...prev,
                                      [parlay.id]: event.target.value,
                                    }))
                                  }
                                  style={stakeInputStyle}
                                />
                              </label>
  const [placeabilityByParlayId, setPlaceabilityByParlayId] = useState({});  const [placeabilityNotesByParlayId, setPlaceabilityNotesByParlayId] = useState({});
                              <label style={miniLabelStyle}>
                                Placeability
                                <select
                                  value={placeabilityStatus}
                                  onChange={(event) => {
                                    const nextStatus = event.target.value;

                                    setPlaceabilityByParlayId((prev) => ({
                                      ...prev,
                                      [parlay.id]: nextStatus,
                                    }));

                                    handlePlaceabilityDecisionChange(parlay, nextStatus, placeabilityNotes, {
                                      selectedBoostId,
                                      stakeValue,
                                      actualSgpRegrade,
                                    });
                                  }}
                                  style={boostSelectStyle}
                                >
                                  <option value="not_checked">Not checked</option>
                                  <option value="accepted">Accepted by book</option>
                                  <option value="rejected">Rejected by book</option>
                                  <option value="does_not_accept">Book does not accept</option>
                                  <option value="could_not_build">Could not build</option>
                                </select>                              </label>

                              <label style={miniLabelStyle}>
                                Placeability Notes
                                <input
                                  value={placeabilityNotes}
                                  placeholder="Rejected two props / accepted at +425 / etc."
                                  onChange={(event) => {
                                    const nextNotes = event.target.value;

                                    setPlaceabilityNotesByParlayId((prev) => ({
                                      ...prev,
                                      [parlay.id]: nextNotes,
                                    }));

                                    handlePlaceabilityDecisionChange(parlay, placeabilityStatus, nextNotes, {
                                      selectedBoostId,
                                      stakeValue,
                                      actualSgpRegrade,
                                    });
                                  }}
                                  style={{ ...stakeInputStyle, width: 260 }}
                                />
                              </label>
                            </div>

                            <div style={subtleStyle}>
                              Placeability changes autosave even if you do not click Save Candidate.
                            </div>

                            {actualSgpRegrade ? (
                              <div style={actualSgpRegradeGridStyle}>
                                <MetricRow label="Odds Input Mode" value={actualSgpRegrade.actualOddsMode === "cluster" ? "Cluster only" : "Full parlay"} />
                                <MetricRow label="Actual Raw SGP Odds" value={formatAmerican(actualSgpRegrade.actualRawAmerican)} />
                                <MetricRow label="Actual Boosted SGP Odds" value={formatAmerican(actualSgpRegrade.actualBoostedAmerican)} />
                                <MetricRow label="Actual Raw EV %" value={formatPct(actualSgpRegrade.actualRawEvPct)} />
                                <MetricRow label="Actual Boosted EV %" value={formatPct(actualSgpRegrade.actualBoostedEvPct)} />
                                <MetricRow label="Actual Suggested Kelly" value={`$${Number(actualSgpRegrade.actualSuggestedKellyStake || 0).toFixed(2)}`} />
                                <MetricRow label="Actual Raw Kelly" value={`$${Number(actualSgpRegrade.actualRawSuggestedKellyStake || 0).toFixed(2)}`} />
                                <MetricRow label="Actual Boosted Kelly" value={`$${Number(actualSgpRegrade.actualBoostedSuggestedKellyStake || 0).toFixed(2)}`} />
                                <MetricRow label="Actual Stake Used" value={`$${Number(actualSgpRegrade.actualStakeUsed || 0).toFixed(2)}`} />
                                <MetricRow label="Actual Expected $" value={`$${Number(actualSgpRegrade.actualExpectedProfitAtStake || 0).toFixed(2)}`} />
                                <MetricRow label="Actual Grade" value={`${actualSgpRegrade.actualGradeTier} / ${actualSgpRegrade.actualPlayLabel}`} />
                              </div>
                            ) : null}
                          </div>
                        ) : null}

                        {parlay.devigWarning ? (
                          <div style={devigWarningStyle}>
                            <div style={{ fontWeight: 900 }}>Devig method warning</div>
                            <div>{parlay.devigWarning}</div>
                            {parlay.recommendedDevigMethod ? (
                              <div style={devigButtonRowStyle}>
                                <button
                                  type="button"
                                  onClick={() => toggleRecommendedDevig(parlay.id)}
                                  style={devigCalcButtonStyle}
                                >
                                  {showRecommended ? "Hide" : "Calculate"} one-off view with {parlay.recommendedDevigMethodLabel}
                                </button>
                              </div>
                            ) : null}
                          </div>
                        ) : null}

                        {showRecommended && parlay.recommendedDevigSnapshot ? (
                          <div style={recommendedBoxStyle}>
                            <div style={{ fontWeight: 900, marginBottom: 6 }}>
                              One-off {parlay.recommendedDevigSnapshot.methodLabel} reference
                            </div>
                            <div style={metricsGridStyle}>
                              <MetricRow label="Fair Hit %" value={formatPct(parlay.recommendedDevigSnapshot.fairHitProbability)} />
                              <MetricRow label="Boosted EV %" value={formatPct(parlay.recommendedDevigSnapshot.expectedValuePct)} />
                              <MetricRow label="Raw EV %" value={formatPct(parlay.recommendedDevigSnapshot.rawExpectedValuePct)} />
                              <MetricRow label="Boosted Kelly" value={`$${Number(parlay.recommendedDevigSnapshot.boostedSuggestedKellyStake ?? 0).toFixed(2)}`} />
                            </div>
                            <div style={subtleStyle}>Reference only. This does not save or change the parlay.</div>
                          </div>
                        ) : null}

                        <div style={{ marginTop: 8, display: "grid", gap: 6 }}>
                          {parlay.legs?.map((leg, legIdx) => (
                            <div key={`${parlay.id}_${legIdx}`} style={legBreakdownRowStyle}>
                              <div style={legTitleRowStyle}>
                                <span>
                                  • {leg.eventName} — {formatLegSelection(leg)}
                                  {getSavedLegUsage(leg, savedLegUsageMap)?.count ? (
                                    <span style={usedLegBadgeStyle}>
                                      Used {getSavedLegUsage(leg, savedLegUsageMap).count}x
                                    </span>
                                  ) : null}
                                </span>

                                <button
                                  type="button"
                                  onClick={() => onBlockParlayLeg?.(leg)}
                                  style={blockLegButtonStyle}
                                >
                                  Block Leg
                                </button>
                              </div>
                              <div style={legBreakdownMetaStyle}>
                                <label style={inlineLegOddsEditStyle}>
                                  Target Odds
                                  <input
                                    value={
                                      legOddsByParlayId[parlay.id]?.[legIdx] ??
                                      formatAmerican(leg.oddsAmerican)
                                    }
                                    onChange={(event) => {
                                      const nextText = event.target.value;

                                      setLegOddsByParlayId((prev) => ({
                                        ...prev,
                                        [parlay.id]: {
                                          ...(prev[parlay.id] || {}),
                                          [legIdx]: nextText,
                                        },
                                      }));

                                      onUpdateParsedRowOddsFromParlayCard?.({
                                        parsedRowId: leg.targetParsedRowId,
                                        oddsText: nextText,
                                        source: "target",
                                        leg,
                                      });
                                    }}
                                    style={inlineLegOddsInputStyle}
                                  />
                                </label>
                                {" "}at {leg.sportsbook}
                                {" "}• Sharp {formatAmerican(leg.sharpOddsAmerican)} at {leg.sharpSportsbook || "sharp source"}
                                {" "}•{" "}
                                <label style={inlineLegOddsEditStyle}>
                                  Fair Odds
                                  <input
                                    value={
                                      fairOddsByParlayId[parlay.id]?.[legIdx] ??
                                      formatAmerican(leg.fairAmerican)
                                    }
                                    onChange={(event) => {
                                      const nextText = event.target.value;

                                      setFairOddsByParlayId((prev) => ({
                                        ...prev,
                                        [parlay.id]: {
                                          ...(prev[parlay.id] || {}),
                                          [legIdx]: nextText,
                                        },
                                      }));

                                      onUpdateParsedRowOddsFromParlayCard?.({
                                        parsedRowId: leg.sharpParsedRowId,
                                        oddsText: nextText,
                                        source: "fair",
                                        leg,
                                      });
                                    }}
                                    style={inlineLegOddsInputStyle}
                                  />
                                </label>
                                {" "}• Leg EV{" "}
                                <span style={Number(leg.legEvPct) >= 0 ? legBreakdownEvStyle : legBreakdownEvNegativeStyle}>
                                  {formatPct(leg.legEvPct)}
                                </span>
                              </div>
                            </div>
                          )) || null}
                        </div>

                        <div style={metricsGridStyle}>
                          <MetricRow label="Raw Odds" value={formatAmerican(parlay.rawParlayAmerican)} />
                          <MetricRow label="Boosted Odds" value={formatAmerican(displayBoostedAmerican)} />
                          <MetricRow label="Fair Hit %" value={formatPct(parlay.fairHitProbability)} />
                          <MetricRow label="Boosted EV %" value={formatPct(displayExpectedValuePct)} />
                          <MetricRow label="Raw EV %" value={formatPct(actualSgpRegrade?.actualRawEvPct ?? parlay.rawExpectedValuePct)} />
                          <MetricRow label="Avg Leg EV %" value={formatPct(parlay.averageLegEvPct)} />
                          <MetricRow label="Expected $" value={`$${Number(displayExpectedProfitAtStake ?? 0).toFixed(2)}`} />

function repairMojibakeDeep(value) {
  if (typeof value === "string") return repairMojibakeText(value);

  if (Array.isArray(value)) {
    return value.map((item) => repairMojibakeDeep(item));
  }

  if (value && typeof value === "object") {
    return Object.fromEntries(
      Object.entries(value).map(([key, item]) => [key, repairMojibakeDeep(item)])
    );
  }

  return value;
}

function americanToDecimalLocal(value) {
  const odds = Number(value);

  if (!Number.isFinite(odds) || odds === 0) return null;
  if (odds > 0) return 1 + odds / 100;
  return 1 + 100 / Math.abs(odds);
}

function decimalToAmericanLocal(decimal) {
  const value = Number(decimal);

  if (!Number.isFinite(value) || value <= 1) return null;
  if (value >= 2) return Math.round((value - 1) * 100);
  return Math.round(-100 / (value - 1));
}

function applyProfitBoostToDecimalLocal(decimal, boostPct) {
  const value = Number(decimal);
  const boost = Number(boostPct || 0) / 100;

  if (!Number.isFinite(value) || value <= 1) return null;
  return 1 + (value - 1) * (1 + boost);
}

function calculateSuggestedKellyStakeLocal({ bankroll, kellyFraction, winProbability, decimalOdds }) {
  const p = Number(winProbability);
  const d = Number(decimalOdds);
  const b = d - 1;
  const q = 1 - p;

  if (!Number.isFinite(bankroll) || bankroll <= 0) return 0;
  if (!Number.isFinite(kellyFraction) || kellyFraction <= 0) return 0;
  if (!Number.isFinite(p) || !Number.isFinite(d) || b <= 0) return 0;

  const fullKelly = (b * p - q) / b;
  if (!(fullKelly > 0)) return 0;

  return bankroll * fullKelly * kellyFraction;
}

function buildActualSgpRegrade(parlay, rawActualAmerican, stakeOverride, options = {}) {
  const enteredActualDecimal = americanToDecimalLocal(rawActualAmerican);
  const fairHitProbability = Number(parlay?.fairHitProbability);
  const boostPct = Number(parlay?.boostPctUsed || 0);
  const enteredStakeText = String(stakeOverride ?? "").trim();
  const enteredStake = Number(enteredStakeText);
  const originalExpectedProfit = Number(parlay?.expectedProfitAtStake);
  const originalExpectedValuePct = Number(parlay?.expectedValuePct);
  const impliedOriginalStake =
    Number.isFinite(originalExpectedProfit) &&
    Number.isFinite(originalExpectedValuePct) &&
    Math.abs(originalExpectedValuePct) > 0.000001
      ? originalExpectedProfit / originalExpectedValuePct
      : 0;
  const stake =
    enteredStakeText !== "" && Number.isFinite(enteredStake)
      ? enteredStake
      : Number(parlay?.stake || impliedOriginalStake || 0);

  if (
    !Number.isFinite(enteredActualDecimal) ||
    !Number.isFinite(fairHitProbability) ||
    fairHitProbability <= 0 ||
    fairHitProbability >= 1
  ) {
    return null;
  }

  const clusterInfo = getSameGameClusterInfo(parlay);
  const requestedMode = String(options.sgpOddsMode || "").trim().toLowerCase();
  const actualOddsMode =
    clusterInfo.isMixedCluster && requestedMode !== "full"
      ? "cluster"
      : "full";

  const actualRawDecimal =
    actualOddsMode === "cluster"
      ? enteredActualDecimal * clusterInfo.remainingDecimal
      : enteredActualDecimal;

  const actualBoostedDecimal = applyProfitBoostToDecimalLocal(actualRawDecimal, boostPct);
  const actualRawAmerican = decimalToAmericanLocal(actualRawDecimal);
  const actualBoostedAmerican = decimalToAmericanLocal(actualBoostedDecimal);

  const actualRawEvPct =
    fairHitProbability * (actualRawDecimal - 1) - (1 - fairHitProbability);

  const actualBoostedEvPct =
    fairHitProbability * (actualBoostedDecimal - 1) - (1 - fairHitProbability);

  const bankroll =
    Number(parlay?.kellyBankrollUsed) ||
    Number(parlay?.bankroll) ||
    0;

  const kellyFraction =
    Number(parlay?.kellyFractionUsed) ||
    Number(parlay?.kellyFraction) ||
    0;

  const actualRawSuggestedKellyStake = calculateSuggestedKellyStakeLocal({
    bankroll,
    kellyFraction,
    winProbability: fairHitProbability,
    decimalOdds: actualRawDecimal,
  });

  const actualBoostedSuggestedKellyStake = calculateSuggestedKellyStakeLocal({
    bankroll,
    kellyFraction,
    winProbability: fairHitProbability,
    decimalOdds: actualBoostedDecimal,
  });

  const actualGrade = gradeActualSgpRegradeLocal({
    parlay,
    actualRawEvPct,
    actualBoostedEvPct,
    actualBoostedAmerican,
  });

  return {
    actualOddsMode,
    actualEnteredAmerican: decimalToAmericanLocal(enteredActualDecimal),
    actualEnteredDecimal: enteredActualDecimal,
    actualRawDecimal,
    actualBoostedDecimal,
    actualRawAmerican,
    actualBoostedAmerican,
    actualRawEvPct,
    actualBoostedEvPct,
    actualStakeUsed: stake,
    actualExpectedProfitAtStake: stake * actualBoostedEvPct,
    actualRawSuggestedKellyStake,
    actualBoostedSuggestedKellyStake,
    actualSuggestedKellyStake: actualBoostedSuggestedKellyStake,
    actualGradeTier: actualGrade.tier,
    actualPlayLabel: actualGrade.label,
    actualGradeScore: actualGrade.score,
    sgpClusterEventName: clusterInfo.eventName,
    sgpClusterLegCount: clusterInfo.clusterLegs.length,
    sgpRemainingLegCount: clusterInfo.remainingLegs.length,
    sgpRemainingDecimal: clusterInfo.remainingDecimal,
  };
}

function getSameGameClusterInfo(parlay = {}) {
  const legs = Array.isArray(parlay.legs) ? parlay.legs : [];
  const eventMap = new Map();

  for (const leg of legs) {
    const key = String(leg.eventName || "").trim();
    if (!key) continue;
    if (!eventMap.has(key)) eventMap.set(key, []);
    eventMap.get(key).push(leg);
  }

  let eventName = "";
  let clusterLegs = [];

  for (const [key, eventLegs] of eventMap.entries()) {
    if (eventLegs.length > clusterLegs.length) {
      eventName = key;
      clusterLegs = eventLegs;
    }
  }

  const hasCluster = clusterLegs.length >= 2;
  const remainingLegs = hasCluster
    ? legs.filter((leg) => String(leg.eventName || "").trim() !== eventName)
    : [];

  const remainingDecimal = remainingLegs.reduce((acc, leg) => {
    const decimal = Number(leg.oddsDecimal);
    return Number.isFinite(decimal) && decimal > 1 ? acc * decimal : acc;
  }, 1);

  return {
    hasCluster,
    isMixedCluster: hasCluster && remainingLegs.length > 0,
    eventName,
    clusterLegs,
    remainingLegs,
    remainingDecimal,
  };
}

function gradeActualSgpRegradeLocal({
  parlay,
  actualRawEvPct,
  actualBoostedEvPct,
  actualBoostedAmerican,
}) {
  const fairHitProbability = Number(parlay?.fairHitProbability || 0);
  const averageLegEvPct = Number(parlay?.averageLegEvPct || 0);
  const hasNegativeLegEv = parlay?.hasNegativeLegEv === true;
  const legCount = Array.isArray(parlay?.legs) ? parlay.legs.length : 0;
  const correlationRisk = String(parlay?.correlationRisk || "low");
  const boostOnly = actualRawEvPct <= 0 || hasNegativeLegEv || averageLegEvPct < 0;
  let score = 0;

  score += clampLocal(actualBoostedEvPct / 0.12, -1, 2) * 35;
  score += clampLocal(Math.max(actualRawEvPct, 0) / 0.06, 0, 1.5) * 20;
  score += clampLocal(fairHitProbability / 0.25, 0, 1.6) * 25;

  if (actualBoostedAmerican >= 1500) score -= 10;
  else if (actualBoostedAmerican >= 800) score -= 5;
  else if (actualBoostedAmerican >= 400) score -= 3;

  score -= Math.max(0, legCount - 2) * 6;

  if (correlationRisk === "high") score -= 12;
  if (correlationRisk === "medium") score -= 6;
  if (hasNegativeLegEv) score -= 12;
  if (averageLegEvPct < 0) score -= 12;
  if (actualRawEvPct <= 0) score -= 14;

  if (boostOnly) {
    if (score >= 44 && actualBoostedEvPct >= 0.06 && fairHitProbability >= 0.15) {
      return { tier: "B", label: "Boost-Only Edge", score };
    }

    if (score >= 30 && actualBoostedEvPct > 0) {
      return { tier: "C", label: "Thin Boost Edge", score };
    }

    return { tier: "D", label: "Promo-Only Longshot", score };
  }

  if (score >= 64 && actualBoostedEvPct >= 0.06 && fairHitProbability >= 0.18) {
    return { tier: "A+", label: "Premium Play", score };
  }

  if (score >= 52 && actualBoostedEvPct >= 0.04 && fairHitProbability >= 0.12) {
    return { tier: "A", label: "Strong Play", score };
  }

  if (score >= 40 && actualBoostedEvPct >= 0.025) {
    return { tier: "B", label: "Core Play", score };
  }

  if (score >= 28 && actualBoostedEvPct > 0) {
    return { tier: "C", label: "Thin Edge", score };
  }

  return { tier: "D", label: "Longshot Value", score };
}

}

function convertOdds(value) {
  const text = String(value || "").trim();
  if (!text) return null;

  if (/^[+-]\d+/.test(text)) {
    const american = Number(text);
    if (!Number.isFinite(american)) return null;
    const decimal = american > 0 ? 1 + american / 100 : 1 + 100 / Math.abs(american);
    return { type: "american_to_decimal", american, decimal };
  }

  if (/^\d+(\.\d+)?$/.test(text)) {
    const decimal = Number(text);
    if (!Number.isFinite(decimal) || decimal <= 1) return null;
    const american = decimal >= 2 ? (decimal - 1) * 100 : -100 / (decimal - 1);
    return { type: "decimal_to_american", decimal, american };
  }

  return null;
}
