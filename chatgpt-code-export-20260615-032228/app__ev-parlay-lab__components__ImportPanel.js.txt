"use client";

import { useState } from "react";

const SPORTSBOOK_OPTIONS = [
  "Auto",
  "DraftKings",
  "FanDuel",
  "BetMGM",
  "Caesars",
  "Pinnacle",
  "TheScore",
  "Manual",
];

export default function ImportPanel({
  rawText,
  setRawText,
  sportsbook,
  setSportsbook,
  batchRole,
  setBatchRole,
  onParse,
  onClearInput,
  onClearParsedRows,
  hasRows,
  lastParsedAt,
  pendingImports = [],
  pendingUrlImport = null,
  onLoadNewestImport,
  onClearPendingImports,
  onClearSavedSession,
  importMode = "append",
  setImportMode,
  defaultCollapsed = true,
}) {
  const [betMgmLadderEvent, setBetMgmLadderEvent] = useState("");
  const [betMgmLadderMarket, setBetMgmLadderMarket] = useState("Player points");
  const [betMgmLadderThreshold, setBetMgmLadderThreshold] = useState("1+");

  function insertBetMgmLadderMarker() {
    const event = String(betMgmLadderEvent || "").trim();
    const market = String(betMgmLadderMarket || "").trim();
    const threshold = String(betMgmLadderThreshold || "").trim();

    if (!event || !market || !threshold) {
      window.alert("Enter event, market, and threshold before inserting the BetMGM ladder marker.");
      return;
    }

    const marker = `BETMGM_MANUAL_LADDER_CAPTURE: sport=WNBA | event=${event} | market=${market} | threshold=${threshold}`;
    const current = String(rawText || "");
    const nextText = current.trim()
      ? `${current.trim()}\n\n${marker}\n`
      : `${marker}\n`;

    setRawText(nextText);
  }

  return (
    <details style={sectionStyle}>
      <summary style={summaryStyle}>
        <span>1. Import Odds</span>
        <span style={summaryMetaStyle}>
          {pendingImports.length ? `${pendingImports.length} pending import${pendingImports.length === 1 ? "" : "s"}` : "Collapsed by default"}
        </span>
      </summary>

      <p style={mutedStyle}>
        Paste extracted odds text here and choose the source book. Extension imports can be loaded from the pending-import tools at the bottom of this box.
      </p>

      <div style={controlsRowStyle}>
        <label style={fieldStyle}>
          <span style={fieldLabelStyle}>Sportsbook</span>
          <select
            value={sportsbook}
            onChange={(e) => {
              const nextBook = e.target.value;
              setSportsbook(nextBook);

              if (nextBook === "Pinnacle") {
                setBatchRole("fair_odds");
              } else if (nextBook !== "Auto") {
                setBatchRole("target");
              }
            }}
            style={inputStyle}
          >
            {SPORTSBOOK_OPTIONS.map((option) => (
              <option key={option} value={option}>
                {option}
              </option>
            ))}
          </select>
        </label>

        <label style={fieldStyle}>
          <span style={fieldLabelStyle}>Batch Role</span>
          <select
            value={batchRole}
            onChange={(e) => setBatchRole(e.target.value)}
            style={inputStyle}
          >
            <option value="target">Target book</option>
            <option value="fair_odds">Fair odds / sharp source</option>
          </select>
        </label>
      </div>

      <div style={betMgmLadderHelperStyle}>
        <div style={betMgmLadderTitleStyle}>BetMGM Manual Ladder Helper</div>
        <div style={betMgmLadderSubtleStyle}>
          Use this when BetMGM shows ladder-only WNBA props. Insert a marker, then paste the ladder text below it.
        </div>

        <div style={betMgmLadderGridStyle}>
          <label style={fieldStyle}>
            <span style={fieldLabelStyle}>Event</span>
            <input
              value={betMgmLadderEvent}
              onChange={(event) => setBetMgmLadderEvent(event.target.value)}
              placeholder="Connecticut Sun @ Chicago Sky"
              style={inputStyle}
            />
          </label>

          <label style={fieldStyle}>
            <span style={fieldLabelStyle}>Market</span>
            <select
              value={betMgmLadderMarket}
              onChange={(event) => setBetMgmLadderMarket(event.target.value)}
              style={inputStyle}
            >
              <option value="Player points">Player points</option>
              <option value="Player rebounds">Player rebounds</option>
              <option value="Player assists">Player assists</option>
              <option value="Player three-pointers">Player three-pointers</option>
              <option value="Player points + rebounds + assists">Player points + rebounds + assists</option>
              <option value="Player points + rebounds">Player points + rebounds</option>
              <option value="Player points + assists">Player points + assists</option>
              <option value="Player rebounds + assists">Player rebounds + assists</option>
            </select>
          </label>

          <label style={fieldStyle}>
            <span style={fieldLabelStyle}>Active Threshold</span>
            <input
              value={betMgmLadderThreshold}
              onChange={(event) => setBetMgmLadderThreshold(event.target.value)}
              placeholder="1+, 2+, 10+, etc."
              style={inputStyle}
            />
          </label>

          <button type="button" onClick={insertBetMgmLadderMarker} style={secondaryButtonStyle}>
            Insert BetMGM Ladder Marker
          </button>
        </div>
      </div>

      <textarea
        value={rawText || ""}
        onChange={(e) => setRawText(e.target.value)}
        style={textareaStyle}
        placeholder="Paste odds text here..."
        spellCheck={false}
      />

      <div style={{ marginTop: 8, fontSize: 12, color: "#166534", fontWeight: 700 }}>
        Input chars: {(rawText || "").length}
      </div>

      <div style={actionRowStyle}>
        <button type="button" onClick={onParse} style={primaryButtonStyle}>
          Parse Input
        </button>

        <button type="button" onClick={onClearInput} style={secondaryButtonStyle}>
          Clear Input
        </button>

        <button
          type="button"
          onClick={onClearParsedRows}
          style={{
            ...dangerButtonStyle,
            ...(hasRows ? {} : disabledButtonStyle),
          }}
          disabled={!hasRows}
        >
          Clear Parsed Rows
        </button>

        <span style={mutedStyle}>
          {lastParsedAt ? `Last parsed: ${new Date(lastParsedAt).toLocaleString()}` : "Not parsed yet"}
        </span>
      </div>

      {pendingUrlImport ? (
        <div style={importNoticeStyle}>
          Imported scraped text from URL. Review or click Parse.
        </div>
      ) : null}

      <div style={pendingPanelStyle}>
        <div style={pendingTitleStyle}>Pending scraped imports: {pendingImports.length}</div>

        <div style={pendingActionRowStyle}>
          <button
            type="button"
            onClick={() => onLoadNewestImport?.({ append: false })}
            style={primaryButtonStyle}
          >
            Load newest import
          </button>

          <button
            type="button"
            onClick={() => onLoadNewestImport?.({ append: true })}
            style={secondaryButtonStyle}
          >
            Append newest import
          </button>

          <label style={inlineFieldStyle}>
            Import Mode
            <select
              value={importMode}
              onChange={(event) => setImportMode?.(event.target.value)}
              style={inputStyle}
            >
              <option value="append">Append / dedupe</option>
              <option value="replace_book">Replace this book</option>
              <option value="replace_book_event">Replace this book + event</option>
            </select>
          </label>

          <button type="button" onClick={onClearPendingImports} style={dangerButtonStyle}>
            Clear pending imports
          </button>

          <button type="button" onClick={onClearSavedSession} style={warningButtonStyle}>
            Clear saved session
          </button>
        </div>
      </div>
    </details>
  );
}
